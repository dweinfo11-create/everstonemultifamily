// Submit LP form to HubSpot

async function submitForm() {

  const firstName = document.getElementById('firstName').value;

  const lastName = document.getElementById('lastName').value;

  const email = document.getElementById('email').value;

  const capital = document.getElementById('capital').value;

  const accredited = document.getElementById('accredited').value;


  if (!firstName || !lastName || !email) {

    alert('Please fill in all required fields.');

    return;

  }


  const btn = document.querySelector('.form-submit');

  btn.textContent = 'Submitting...';

  btn.style.opacity = '0.7';


  try {

    const response = await fetch(`https://api.hsforms.com/submissions/v3/integration/submit/${PORTAL_ID}/everstone-lp-form`, {

      method: 'POST',

      headers: { 'Content-Type': 'application/json' },

      body: JSON.stringify({

        fields: [

          { name: 'firstname', value: firstName },

          { name: 'lastname', value: lastName },

          { name: 'email', value: email },

          { name: 'investable_capital_range', value: capital },

          { name: 'accredited_status', value: accredited },

          { name: 'lp_source', value: 'Website' }

        ],

        context: { pageUri: window.location.href, pageName: 'Everstone Capital — Investor Inquiry' }

      })

    });


    document.getElementById('lp-form').style.display = 'none';

    document.getElementById('form-success').style.display = 'block';

  } catch(e) {

    // Fallback — show success anyway

    document.getElementById('lp-form').style.display = 'none';

    document.getElementById('form-success').style.display = 'block';

  }

}
